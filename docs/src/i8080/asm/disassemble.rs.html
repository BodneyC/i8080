1//! 8080 disassembler
2//!
3//! This is much simpler than the assembler, it reads a stream of `u8`s from a file and interprets
4//! them as 8080 instructions.
5//!
6//! This does nothing to try to "intelligently" resolve labels, macros, strings-literals, etc. For
7//! example,
8//!
9//! ```asm
10//! DB 'hello world'
11//! ```
12//!
13//! would be disassembled as:
14//!
15//! ```asm
16//! MOV L, B
17//! MOV H, L
18//! MOV L, H
19//! MOV L, H
20//! MOV L, A
21//! ---
22//! MOV M, A
23//! MOV L, A
24//! MOV M, D
25//! MOV L, H
26//! MOV H, H
27//! ```
28//!
29//! # Examples
30//!
31//! To disassemble a file to a specific output file
32//!
33//! ```sh
34//! $ i8080 dis rsc/bin/simple.bin
35//! MVI A, 0xde
36//! MVI B, 0xad
37//! ADD B
38//! HLT
39//! ```

46/// Disassemble the instruction at `from`, returning its text and width in bytes
47pub fn disassemble_instruction(v: &[u8], from: usize) -> Result<(String, usize), DisassembleError> {
48    let v = v.get(from..).unwrap_or_default();
49    let inst = *v.first().ok_or(DisassembleError::NoRemainingBytes(from))?;
50    let meta = I8080_OP_META[usize::from(inst)];
51    if v.len() < meta.width() {
52        return Err(DisassembleError::NotEnoughBytes(from, meta.op, v.to_vec()));
53    }
54    let mut op: String = meta.op.to_owned();
55    if meta.argb || meta.argw {
56        if meta.asm_arg_count == 2 {
57            op.push(',');
58        }
59        if meta.argb {
60            op.push_str(&format!(" {:#04x}", v[1]));
61        } else {
62            op.push_str(&format!(" {:#06x}", vec_u8_to_u16(&v[1..3])));
63        }
64    }
65    Ok((op, meta.width()))
66}
67
68pub fn disassemble_vec(v: &[u8]) -> Result<Vec<String>, DisassembleError> {
69    let mut strings = Vec::new();
70    let mut from = 0;
71    while from < v.len() {
72        let (s, width) = disassemble_instruction(v, from)?;
73        strings.push(s);
74        from += width;
75    }
76    Ok(strings)
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    #[test]
84    fn instructions_with_arguments() {
85        let v = [0x3e, 0xde, 0x01, 0x34, 0x12, 0xc3, 0xef, 0xbe];
86        assert_eq!(
87            disassemble_vec(&v).unwrap(),
88            vec!["MVI A, 0xde", "LXI B, 0x1234", "JMP 0xbeef"]
89        );
90    }
91
92    #[test]
93    fn truncated_input() {
94        assert!(matches!(
95            disassemble_vec(&[0xc3, 0x00]),
96            Err(DisassembleError::NotEnoughBytes(0, "JMP", _))
97        ));
98        assert!(matches!(
99            disassemble_instruction(&[0x00], 5),
100            Err(DisassembleError::NoRemainingBytes(5))
101        ));
102    }
103}
