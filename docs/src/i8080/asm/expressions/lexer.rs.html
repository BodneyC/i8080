10/// Properties of an expression which affect how the assembler treats its value
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
12pub struct ExprFlags {
13    /// The expression is the `SP` register
14    pub sp: bool,
15    /// The expression is the `PSW` register pair
16    pub psw: bool,
17    /// The expression uses the program counter (`$`)
18    pub pc: bool,
19    /// The expression is a string of more than one character
20    pub string: bool,
21}
22
23impl ExprFlags {
24    pub fn new() -> Self {
25        Self::default()
26    }
27}

29fn functions(ident: &str) -> Option<Token> {
30    match ident {
31        "XOR" => Some(Token::Operator('^', Assoc::Left, 4)),
32        "AND" => Some(Token::Operator('&', Assoc::Left, 4)),
33        "OR" => Some(Token::Operator('|', Assoc::Left, 4)),
34        "NOT" | "NEG" => Some(Token::Unary(ident.to_string())),
35        _ => None,
36    }
37}

39pub struct Lexer<'a> {
40    iter: iter::Peekable<str::Chars<'a>>,
41    address: u16,
42}
43
44impl<'a> Lexer<'a> {
45    pub fn new() -> Self {
46        Self {
47            iter: "".chars().peekable(),
48            address: 0,
49        }
50    }
51
52    pub fn lex(
53        &mut self,
54        raw_str: &'a str,
55        address: u16,
56        labels: &HashMap<String, Label>,
57    ) -> Result<(Vec<Token>, ExprFlags), ExpressionError> {
58        self.iter = raw_str.chars().peekable();
59        self.address = address;
60
61        let mut flags: ExprFlags = ExprFlags::new();
62        let mut tokens: Vec<Token> = Vec::new();
63
64        // Should we skip advancing if a sub method has done it for us?
65        while let Some(&c) = self.iter.peek() {
66            let consume_next: bool = if c.is_whitespace() {
67                true
68            } else if c.is_ascii_digit() {
69                let (number, radix) = self.consume_number();
70                let val = u16::from_str_radix(&number, radix)
71                    .map_err(ExpressionError::NumberParseError)?;
72                tokens.push(Token::Number(val));
73                false
74            } else if c.is_alphabetic() || c == '_' {
75                let ident = self.consume_identifier();
76                if let Some(operator) = functions(&ident) {
77                    tokens.push(operator);
78                }
79                // These should come before the label check as someone might use an SP label
80                else if ident == "PSW" {
81                    flags.psw = true;
82                    tokens.push(Token::MetaIdentifier(ident));
83                } else if ident == "SP" {
84                    flags.sp = true;
85                    tokens.push(Token::MetaIdentifier(ident));
86                } else if let Some(label) = labels.get(&ident) {
87                    tokens.push(Token::Number(label.value));
88                } else {
89                    return Err(ExpressionError::UnknownIdentifier(ident));
90                }
91                false
92            } else {
93                match c {
94                    '\'' => {
95                        let s = self.consume_string()?;
96                        let mut chars = s.chars();
97                        if let (Some(ch), None) = (chars.next(), chars.next()) {
98                            tokens.push(Token::Number(ch as u16))
99                        } else {
100                            flags.string = true;
101                            tokens.push(Token::String(s));
102                        }
103                    }
104                    '$' => {
105                        flags.pc = true;
106                        tokens.push(Token::Number(self.address));
107                    }
108                    '+' | '-' => {
109                        tokens.push(Token::Operator(c, Assoc::Left, 2));
110                    }
111                    '*' | '/' => {
112                        tokens.push(Token::Operator(c, Assoc::Left, 3));
113                    }
114                    '(' => tokens.push(Token::LParen),
115                    ')' => tokens.push(Token::RParen),
116                    _ => return Err(ExpressionError::UnprocessableChar(c)),
117                }
118                true
119            };
120
121            if consume_next {
122                self.iter.next();
123            }
124        }
125
126        Ok((tokens, flags))
127    }
128
129    fn consume_string(&mut self) -> Result<String, ExpressionError> {
130        self.iter.next(); // Consume '\''
131
132        let mut escaped = false;
133        let mut quote_matched = false;
134
135        let mut s = String::new();
136
137        for c in self.iter.by_ref() {
138            if escaped {
139                escaped = false;
140                let escaped_char: char = match c {
141                    '\\' => '\\',
142                    '\'' => '\'',
143                    'n' => '\n',
144                    'r' => '\r',
145                    't' => '\t',
146                    _ => return Err(ExpressionError::UnknownEscape(c)),
147                };
148                s.push(escaped_char);
149            } else {
150                if c == '\'' {
151                    quote_matched = true;
152                    break;
153                }
154                if c == '\\' {
155                    escaped = true;
156                } else {
157                    s.push(c);
158                }
159            }
160        }
161
162        if quote_matched {
163            Ok(s)
164        } else {
165            Err(ExpressionError::UnmatchedQuote(s))
166        }
167    }
168
169    fn consume_number(&mut self) -> (String, u32) {
170        let mut s = String::new();
171        let mut radix: u32 = 10;
172        while let Some(&c) = self.iter.peek() {
173            if c.is_ascii_digit() || (radix == 16 && ('A'..='F').contains(&c)) {
174                s.push(c);
175            } else if s.len() == 1 && s.starts_with('0') {
176                if c == 'X' {
177                    radix = 16;
178                } else if c == 'B' {
179                    radix = 2;
180                } else if c == 'O' {
181                    radix = 8;
182                } else {
183                    break;
184                }
185            } else {
186                break;
187            }
188            self.iter.next();
189        }
190        (s, radix)
191    }
192
193    // Consumes an identifier until we don't have any other letters available
194    fn consume_identifier(&mut self) -> String {
195        let mut s = String::new();
196        while let Some(&c) = self.iter.peek() {
197            if c.is_alphanumeric() || c == '_' {
198                s.push(c);
199            } else {
200                break;
201            }
202            self.iter.next();
203        }
204        s
205    }

219    fn identifiers() {
220        let mut lexer = Lexer::new();
221
222        let s = "_ident ident __id__ent__ other";
223        lexer.set_input(s, 0);
224
225        let out = lexer.consume_identifier();
226        assert_eq!(out, "_ident".to_string());
227
228        lexer.iter.next(); // Whitespace
229
230        let out = lexer.consume_identifier();
231        assert_eq!(out, "ident".to_string());
232
233        lexer.iter.next(); // Whitespace
234
235        let out = lexer.consume_identifier();
236        assert_eq!(out, "__id__ent__".to_string());
237
238        let next = lexer.iter.next(); // Whitespace
239
240        assert!(next.is_some(), "' other' should remain");
241        assert_eq!(next.unwrap(), ' ', "' ' is next");
242
243        let s = "_ident8080";
244        lexer.set_input(s, 0);
245
246        let out = lexer.consume_identifier();
247        assert_eq!(out, "_ident8080".to_string());
248
249        let next = lexer.iter.next();
250
251        assert!(next.is_none(), "numerics consumed too");
252    }
