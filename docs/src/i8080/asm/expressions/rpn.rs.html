3pub fn calculate(input_queue: &[Token]) -> Result<u16, ExpressionError> {
4    trace!("rpn input queue: {:?}", input_queue);
5
6    let mut stack: Vec<u16> = Vec::new();
7
8    for tok in input_queue {
9        match tok {
10            Token::Number(n) => stack.push(*n),
11            Token::Operator(o, _, _) => {
12                let (Some(right), Some(left)) = (stack.pop(), stack.pop()) else {
13                    return Err(ExpressionError::MissingOperand(*o));
14                };
15                stack.push(operate(*o, left, right)?);
16            }
17            Token::Unary(unary) => {
18                let arg = stack.pop().ok_or_else(|| {
19                    ExpressionError::CalculationError(format!("no argument for {}", unary))
20                })?;
21                match unary.as_str() {
22                    "NOT" => stack.push(!arg),
23                    "NEG" => stack.push(arg.wrapping_neg()),
24                    _ => return Err(ExpressionError::UnknownUnary(unary.to_string())),
25                }
26            }
27            tok => return Err(ExpressionError::NotANumber(tok.clone())),
28        }
29    }
30
31    match stack.pop() {
32        Some(n) => {
33            trace!("rpn result: {:#04x}", n);
34            Ok(n)
35        }
36        None => Err(ExpressionError::CalculationError(
37            "empty expression".to_string(),
38        )),
39    }
40}
41
42fn operate(operator: char, left: u16, right: u16) -> Result<u16, ExpressionError> {
43    Ok(match operator {
44        '+' => left.wrapping_add(right),
45        '-' => left.wrapping_sub(right),
46        '*' => left.wrapping_mul(right),
47        '/' => left
48            .checked_div(right)
49            .ok_or(ExpressionError::DivisionByZero)?,
50        '&' => left & right,
51        '^' => left ^ right,
52        '|' => left | right,
53        _ => return Err(ExpressionError::UnprocessableChar(operator)),
54    })
55}
