1//! Find the operation code for a given string
2//!
3//! Depending on what part of the assembly process we're in we may not be able to resolve the true
4//! operation code.
5//!
6//! Say we have `MOV A, _label`, without knowing what that `_label`'s value is, we cannot know the
7//! true op-code.
8//!
9//! We can get some information, all `MOV X, Y` instructions have the same metadata, so depending
10//! on the arguments provided to these methods, you will either get an op-code estimation or the
11//! real thing.
12//!
13//! Then, when labels have been resolved, we can use the same method to find the real code.
14
15use crate::meta::I8080_OP_META;
16
17use super::errors::OpParseError;

19/// Estimate the op-code of an instruction before its arguments are known
20pub fn from_args(inst: &str, arg0: u16, arg1: u16) -> Result<usize, OpParseError> {
21    from_args_and_sp_psw(inst, arg0, arg1, false, false)
22}
23
24/// Find the op-code of an instruction from its resolved arguments
25///
26/// `sp` and `psw` indicate the first argument was the `SP` or `PSW` register respectively
27pub fn from_args_and_sp_psw(
28    inst: &str,
29    arg0: u16,
30    arg1: u16,
31    sp: bool,
32    psw: bool,
33) -> Result<usize, OpParseError> {
34    let extra = sp || psw;
35    match inst {
36        "MOV" => match with_all_regs(0x40, arg0, arg1, extra)? {
37            0x76 => Err(OpParseError::MovAsHalt),
38            code => Ok(code),
39        },
40        "MVI" => with_all_regs(0x06, arg0, 0, extra),
41        "ADD" => with_all_regs(0x80, 0, arg0, extra),
42        "ADC" => with_all_regs(0x88, 0, arg0, extra),
43        "SUB" => with_all_regs(0x90, 0, arg0, extra),
44        "SBB" => with_all_regs(0x98, 0, arg0, extra),
45        "ANA" => with_all_regs(0xa0, 0, arg0, extra),
46        "XRA" => with_all_regs(0xa8, 0, arg0, extra),
47        "ORA" => with_all_regs(0xb0, 0, arg0, extra),
48        "CMP" => with_all_regs(0xb8, 0, arg0, extra),
49        "INR" => with_all_regs(0x04, arg0, 0, extra),
50        "DCR" => with_all_regs(0x05, arg0, 0, extra),
51        "RST" => with_all_regs(0xc7, arg0, 0, extra),
52
53        "LXI" | "DAD" | "INX" | "DCX" if psw => Err(OpParseError::InvalidRegister),
54        "LXI" => with_word_regs(0x01, arg0, sp),
55        "DAD" => with_word_regs(0x09, arg0, sp),
56        "INX" => with_word_regs(0x03, arg0, sp),
57        "DCX" => with_word_regs(0x0b, arg0, sp),
58
59        "PUSH" | "POP" if sp => Err(OpParseError::InvalidRegister),
60        "PUSH" => with_word_regs(0xc5, arg0, psw),
61        "POP" => with_word_regs(0xc1, arg0, psw),
62
63        "LDAX" => with_b_or_d(0x0a, arg0, extra),
64        "STAX" => with_b_or_d(0x02, arg0, extra),
65
66        _ => I8080_OP_META
67            .into_iter()
68            .position(|meta| inst == meta.op)
69            .ok_or_else(|| OpParseError::NoSuchInstruction(inst.to_string())),
70    }
71}

73fn with_all_regs(start: u16, arg0: u16, arg1: u16, extra: bool) -> Result<usize, OpParseError> {
74    if extra || arg0 > 7 || arg1 > 7 {
75        Err(OpParseError::UnknownRegister)
76    } else {
77        let idx = start + (arg0 * 8) + arg1;
78        Ok(idx as usize)
79    }
80}
81
82/// Register pair B (0) or D (2)
83fn with_b_or_d(start: u16, arg0: u16, extra: bool) -> Result<usize, OpParseError> {
84    match arg0 {
85        0 | 2 if !extra => Ok((start + (arg0 / 2) * 0x10) as usize),
86        _ => Err(OpParseError::InvalidRegister),
87    }
88}
89
90/// Register pair B (0), D (2), H (4), or SP/PSW (`extra`)
91fn with_word_regs(start: u16, arg0: u16, extra: bool) -> Result<usize, OpParseError> {
92    let offset = match arg0 {
93        _ if extra => 3,
94        0 | 2 | 4 => arg0 / 2,
95        _ => return Err(OpParseError::InvalidRegister),
96    };
97    Ok((start + offset * 0x10) as usize)
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    #[test]
105    fn register_pair_instructions() {
106        assert_eq!(from_args("LDAX", 0, 0).unwrap(), 0x0a);
107        assert_eq!(from_args("LDAX", 2, 0).unwrap(), 0x1a);
108        assert_eq!(from_args("STAX", 0, 0).unwrap(), 0x02);
109        assert_eq!(from_args("STAX", 2, 0).unwrap(), 0x12);
110        assert!(from_args("LDAX", 4, 0).is_err());
111        assert_eq!(
112            from_args_and_sp_psw("LXI", 0, 0, true, false).unwrap(),
113            0x31
114        );
115        assert_eq!(
116            from_args_and_sp_psw("PUSH", 0, 0, false, true).unwrap(),
117            0xf5
118        );
119        assert!(from_args_and_sp_psw("LXI", 0, 0, false, true).is_err());
120        assert!(from_args_and_sp_psw("PUSH", 0, 0, true, false).is_err());
121    }
122
123    #[test]
124    fn mov_m_m_is_not_halt() {
125        assert!(matches!(
126            from_args("MOV", 6, 6),
127            Err(OpParseError::MovAsHalt)
128        ));
129    }
130
131    #[test]
132    fn named_instructions() {
133        assert_eq!(from_args("RLC", 0, 0).unwrap(), 0x07);
134        assert_eq!(from_args("SHLD", 0, 0).unwrap(), 0x22);
135        assert_eq!(from_args("LHLD", 0, 0).unwrap(), 0x2a);
136        assert!(matches!(
137            from_args("NOPE", 0, 0),
138            Err(OpParseError::NoSuchInstruction(_))
139        ));
140    }
141}
