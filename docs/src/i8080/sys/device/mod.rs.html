1//! Simplistic IO device
2//!
3//! The emulator can use IO devices with its `IN` and `OUT` instructions, so the `IN` will be used
4//! to receive data (`rx`) and `OUT` will be used to transmit (`tx`) data.
5//!
6//! The devices here take to relevant opposites to the above.
7
8pub mod console_device;
9
10use std::sync::mpsc::{Receiver, Sender};
11
12pub struct TxDevice {
13    pub tx: Sender<u8>,
14    pub eot_byte: u8,
15}
16
17impl TxDevice {
18    pub fn new(tx: Sender<u8>, eot_byte: u8) -> Self {
19        Self { tx, eot_byte }
20    }
21}
22
23pub struct RxDevice {
24    pub rx: Receiver<u8>,
25}
26
27impl RxDevice {
28    #[allow(dead_code)] // This may be used at some point...
29    pub fn new(rx: Receiver<u8>) -> Self {
30        Self { rx }
31    }
32}
