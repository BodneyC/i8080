399    /// Writes a byte (usually A) to the specified out-device
400    ///
401    /// If no such in-device exists, nothing is written
402    ///
403    /// # Arguments
404    ///
405    /// - val: Value to pass to the out-device
406    pub fn dev_out(&self, device: usize, val: u8) {
407        match self.tx_devices.get(device) {
408            Some(dev) => {
409                if let Err(err) = dev.tx.send(val) {
410                    debug!("error transmitting {} OUT: {:?}", val, err);
411                }
412            }
413            None => debug!("no device OUT"),
414        }
415    }
416
417    /// Read a byte from the specified in-device and writes it to A
418    ///
419    /// If no such in-device exists, will read all ones, i.e. 0xff
420    fn dev_in(&self, device: usize) -> u8 {
421        match self.rx_devices.get(device).map(|dev| dev.rx.try_recv()) {
422            Some(Ok(val)) => return val,
423            Some(Err(_)) => debug!("nothing received IN"),
424            None => debug!("no device IN"),
425        }
426        0xff
427    }

429    /// Halts the CPU at the end of the current cycle (if using `i8080::run`)
430    ///
431    /// Though not true to the CPU, sends an end of transmission byte to all out-devices.
432    /// This allows the program to end and is akin to a shutdown signal I suppose...
433    /// kinda
434    pub fn halt(&mut self) {
435        self.halted = true;
436        for (idx, device) in self.tx_devices.iter().enumerate() {
437            self.dev_out(idx, device.eot_byte);
438        }
439    }

441    /// Adds a value to the A register
442    ///
443    /// This function takes a `carry` argument. If `false` this functions acts
444    /// like a [ADD A, x] instruction where as if using the real value of the
445    /// `carry` register, this acts like [ADC A, x].
446    ///
447    /// # Arguments
448    ///
449    /// - val: Value to add
450    /// - carry: Carry flag
451    fn add(&mut self, val: u8, carry: bool) {
452        let result16: u16 = (self.registers.a as u16)
453            .wrapping_add(val as u16)
454            .wrapping_add(carry as u16);
455        self.flags.carry = result16 > 0xff;
456        self.flags.aux_carry = (self.registers.a & 0xf)
457            .wrapping_add(val & 0xf)
458            .wrapping_add(carry as u8)
459            > 0xf;
460        self.registers.a = result16 as u8;
461        self.flags.zero_sign_parity(self.registers.a);
462    }
463
464    /// Subtracts a value from the A register
465    ///
466    /// This function takes a `carry` argument. If `false` this functions acts
467    /// like a [SUB A, x] instruction where as if using the real value of the
468    /// `carry` register, this acts like [SBB A, x].
469    ///
470    /// # Arguments
471    ///
472    /// - val: Value to subtract
473    /// - carry: Carry flag
474    fn sub(&mut self, val: u8, carry: bool) {
475        self.flags.carry = (self.registers.a as u16) < (val.wrapping_add(carry as u8)) as u16;
476        self.flags.aux_carry = (self.registers.a as i8 & 0xf)
477            .wrapping_sub(val as i8 & 0xf)
478            .wrapping_sub(carry as i8)
479            >= 0;
480        self.registers.a = ((self.registers.a as u16)
481            .wrapping_sub(val as u16)
482            .wrapping_sub(carry as u16)) as u8;
483        self.flags.zero_sign_parity(self.registers.a);
484    }

486    /// Performs a logical AND between a given value and A
487    ///
488    /// # Arguments
489    ///
490    /// - val: Value to be ANDed with A
491    fn ana(&mut self, val: u8) {
492        self.flags.carry = false;
493        self.flags.aux_carry = ((self.registers.a | val) & 0b1000) != 0;
494        self.registers.a &= val;
495        self.flags.zero_sign_parity(self.registers.a);
496    }

498    /// Performs a logical XOR between a given value and A
499    ///
500    /// # Arguments
501    ///
502    /// - val: Value to be XORed with A
503    fn xra(&mut self, val: u8) {
504        self.registers.a ^= val;
505        self.flags.carry = false;
506        self.flags.aux_carry = false;
507        self.flags.zero_sign_parity(self.registers.a);
508    }
509
510    /// Performs a logical OR between a given value and A
511    ///
512    /// # Arguments
513    ///
514    /// - val: Value to be ORed with A
515    fn ora(&mut self, val: u8) {
516        self.registers.a |= val;
517        self.flags.carry = false;
518        self.flags.aux_carry = false;
519        self.flags.zero_sign_parity(self.registers.a);
520    }
521
522    /// Compares a value against the A register
523    ///
524    /// A CMP instruction is essentially a [SUB A, x] but the result is
525    /// temporary and discarded. Here, for ease, I am replacing A in
526    /// `self.sub(...)` and then restoring it.
527    ///
528    /// # Arguments
529    ///
530    /// - val: Value to compare
531    fn cmp(&mut self, val: u8) {
532        let a: u8 = self.registers.a;
533        self.sub(val, false);
534        self.registers.a = a;
535    }
536
537    /// Increment a value, setting flags
538    ///
539    /// This instruction does not set the carry flag
540    ///
541    /// # Arguments
542    ///
543    /// - val: Value to increment
544    fn inr(&mut self, val: u8) -> u8 {
545        let result = val.wrapping_add(1);
546        self.flags.aux_carry = (result & 0xf) == 0;
547        self.flags.zero_sign_parity(result);
548        result
549    }
550
551    /// Decrement a value, setting flags
552    ///
553    /// This instruction does not set the carry flag
554    ///
555    /// # Arguments
556    ///
557    /// - val: Value to decrement
558    fn dcr(&mut self, val: u8) -> u8 {
559        let result = val.wrapping_sub(1);
560        self.flags.aux_carry = (result & 0xf) != 0xf;
561        self.flags.zero_sign_parity(result);
562        result
563    }
564
565    /// Continues execution at a given address based on a condition
566    ///
567    /// # Arguments
568    ///
569    /// - addr: Address to which PC should be set
570    /// - condition: The condition to test before jumping
571    fn jmp(&mut self, addr: u16, condition: bool) {
572        if condition {
573            self.registers.pc = addr;
574        }
575    }
576
577    /// Push a value onto the stack, stored little endian as on a real 8080
578    fn push(&mut self, val: u16) {
579        self.registers.sp = self.registers.sp.wrapping_sub(2);
580        self.memory.write_word_little_endian(self.registers.sp, val);
581    }
582
583    /// Pop a value from the stack
584    fn pop(&mut self) -> u16 {
585        let result = self.memory.read_word_little_endian(self.registers.sp);
586        self.registers.sp = self.registers.sp.wrapping_add(2);
587        result
588    }
589
590    /// Push A and the flags register to the stack
591    ///
592    /// [A, Flags] -> stack
593    fn push_psw(&mut self) {
594        let val: u16 = (self.registers.a as u16) << 8 | (self.flags.to_byte() as u16);
595        self.push(val);
596    }
597
598    /// Pop A and the flags register from the stack
599    ///
600    /// stack -> [A, Flags]
601    fn pop_psw(&mut self) {
602        let val: u16 = self.pop();
603        self.registers.a = (val >> 8) as u8;
604        self.flags.set_from_byte(val as u8);
605    }
606
607    /// Exchange the value pointed to by the stack pointer with the value of
608    /// the HL register
609    fn xthl(&mut self) {
610        let indirect: u16 = self.memory.read_word_little_endian(self.registers.sp);
611        self.memory
612            .write_word_little_endian(self.registers.sp, self.registers.get_hl());
613        self.registers.set_hl(indirect);
614    }
615
616    /// Exchange the value of HL with DE
617    fn xchg(&mut self) {
618        let hl: u16 = self.registers.get_hl();
619        self.registers.set_hl(self.registers.get_de());
620        self.registers.set_de(hl);
621    }
622
623    /// Whether a conditional call/return is taken, adding the extra cycles taken branches cost
624    fn branch_taken(&mut self, condition: Option<bool>) -> bool {
625        match condition {
626            None => true,
627            Some(taken) => {
628                if taken {
629                    self.cycles += 6;
630                }
631                taken
632            }
633        }
634    }
635
636    /// Push the return address (PC, already advanced) and jump to `addr`
637    fn call(&mut self, addr: u16, condition: Option<bool>) {
638        if self.branch_taken(condition) {
639            self.push(self.registers.pc);
640            self.registers.pc = addr;
641        }
642    }
643
644    fn ret(&mut self, condition: Option<bool>) {
645        if self.branch_taken(condition) {
646            self.registers.pc = self.pop();
647        }
648    }
649
650    /// Decimal Adjust Accumulator, represents the 8-bit value as two, 4-bit,
651    /// decimal numbers
652    fn daa(&mut self) {
653        let mut to_add = 0;
654        let mut carry = self.flags.carry;
655
656        let lo = self.registers.a & 0x0f;
657        if lo > 9 || self.flags.aux_carry {
658            to_add += 0x06;
659        }
660
661        let hi = self.registers.a >> 4;
662        if hi > 9 || self.flags.carry || (hi >= 9 && lo > 9) {
663            to_add += 0x60;
664            carry = true;
665        }
666
667        self.add(to_add, false);
668        self.flags.carry = carry;
669    }
670
671    /// Double add, adds a word (a double) to HL
672    fn dad(&mut self, val: u16) {
673        let hl: u16 = self.registers.get_hl();
674        self.flags.carry = hl > 0xffff_u16.wrapping_sub(val);
675        self.registers.set_hl(hl.wrapping_add(val));
676    }
677
678    /// Rotate A left once, no carry
679    fn rlc(&mut self) {
680        self.flags.carry = util::is_bit_set(self.registers.a, 7);
681        self.registers.a = (self.registers.a << 1) | self.flags.carry as u8;
682    }
683
684    /// Rotate A right once, no carry
685    fn rrc(&mut self) {
686        self.flags.carry = util::is_bit_set(self.registers.a, 0);
687        self.registers.a = if self.flags.carry {
688            0x80 | (self.registers.a >> 1)
689        } else {
690            self.registers.a >> 1
691        };
692    }
693
694    /// Rotate A left once, with carry
695    fn ral(&mut self) {
696        let carry = util::is_bit_set(self.registers.a, 7);
697        self.registers.a = (self.registers.a << 1) | self.flags.carry as u8;
698        self.flags.carry = carry;
699    }
700
701    /// Rotate A right once, with carry
702    fn rar(&mut self) {
703        let carry = util::is_bit_set(self.registers.a, 0);
704        self.registers.a = if self.flags.carry {
705            0x80 | (self.registers.a >> 1)
706        } else {
707            self.registers.a >> 1
708        };
709        self.flags.carry = carry;
710    }

718    fn pop_and_push() {
719        let mut i8080 = I8080::new(vec![], vec![]);
720        i8080.load(
721            0x00,
722            &[
723                0x31, 0xff, 0xff, // LXI SP u16::MAX
724                0x01, 0xad, 0xde, // LXI B 0xdead
725                0xc5, // PUSH B
726                0xd1, // POP D
727            ],
728        );
729        i8080.cycle(); // LXI SP u16::MAX
730        i8080.cycle(); // LXI B 0xdead
731        assert_eq!(i8080.registers.get_bc(), 0xdead, "BC is 0xdead");
732        i8080.cycle(); // PUSH B
733        assert_eq!(i8080.registers.sp, 0xfffd, "SP is initial value - 2");
734        assert_eq!(
735            i8080.memory.read_word_little_endian(i8080.registers.sp),
736            0xdead,
737            "[SP] is 0xdead (little endian)"
738        );
739        i8080.cycle(); // POP D
740        assert_eq!(i8080.registers.sp, 0xffff, "SP is initial value");
741        assert_eq!(i8080.registers.get_de(), 0xdead, "DE is 0xdead");
